]

function makeNode(name, x, y, type='python', code=null) {
  const spec = taskLibrary.find(t=>t.type===type) || taskLibrary[2]
  return {
    id: uid('node'), name, type, x, y, status:'Idle', retries:2, timeout:10,
    simulateError:false, params:{}, inputs:clone(spec.inputs).map(p=>({...p,id:uid('in')})), outputs:clone(spec.outputs).map(p=>({...p,id:uid('out')})),
    code: code || spec.code.replace(/def\s+[a-zA-Z0-9_]+/, 'def '+name), duration:0, retryCount:0
  }
}

function baseWorkflow(id, name, schedule, nodes, edges=[]) {
  return {id,name,schedule,status:'Idle',nodes,edges,undo:[],redo:[],zoom:1,panX:0,panY:0,lastRun:'nigdy'}
}

function ensureFiles(project){
  if(project.files.length) return
  project.workflows.forEach((wf,i)=>project.files.push({id:'f'+(i+1),path:'workflows/'+wf.name+'.py',content:workflowFile(wf),conflict:false,invalid:false,externalContent:null}))
  project.files.push({id:'f_cfg_'+project.id,path:'config/project.json',content:JSON.stringify({name:project.name,visibility:project.visibility},null,2),conflict:false,invalid:false,externalContent:null})
}
